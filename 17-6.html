🥟 包饺子流程 (Promise + async/await)

流程: 买原料(6s) → 取原料(1s) → 并行[和面(4s)、和馅(3s)] → 包饺子(5s) → 吃饺子(2s)

⏱ 实时时间线

📖 核心代码

// Promise 包装 setTimeout
function delay(msg, sec) {
    return new Promise(resolve => {
        setTimeout(() => {
            resolve(msg + ' 完成!');
        }, sec * 1000);
    });
}

// async/await 串行 + Promise.all 并行
async function makeDumplings() {
    await delay('买原料', 6);
    await delay('取原料', 1);
    // 两个任务同时开始
    await Promise.all([
        delay('和面', 4),
        delay('和馅', 3)
    ]);
    await delay('包饺子', 5);
    await delay('吃饺子', 2);
}
